# 1. Assembly fordítása
nasm -f elf32 boot.asm -o boot.o

# 2. C kód fordítása (standard könyvtárak nélkül)
gcc -m32 -c kernel.c -o kernel.o -std=gnu99 -ffreestanding -O2 -Wall

# 3. Összelinkelés egy kisfájllá
ld -m elf_i386 -T linker.ld -o myos.bin boot.o kernel.o

# 4. Futtatás a Virtuális Gépben (QEMU)!
qemu-system-i386 -kernel myos.bin
